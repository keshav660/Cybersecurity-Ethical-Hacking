4.2 Incident Detection Simulation
SIEM Alert Creation
bash
# Create custom detection rules in Elasticsearch

# SSH Bruteforce Detection
PUT _watcher/watch/ssh_bruteforce
{
  "trigger": {
    "schedule": {
      "interval": "1m"
    }
  },
  "input": {
    "search": {
      "request": {
        "indices": ["logs-*"],
        "body": {
          "query": {
            "bool": {
              "must": [
                {
                  "match": {
                    "event.module": "system"
                  }
                },
                {
                  "match": {
                    "system.auth.ssh.event": "Failed"
                  }
                }
              ],
              "filter": {
                "range": {
                  "@timestamp": {
                    "gte": "now-1m"
                  }
                }
              }
            }
          },
          "aggs": {
            "source_ips": {
              "terms": {
                "field": "source.ip",
                "min_doc_count": 5
              }
            }
          }
        }
      }
    }
  },
  "condition": {
    "compare": {
      "ctx.payload.aggregations.source_ips.buckets.0.doc_count": {
        "gte": 5
      }
    }
  },
  "actions": {
    "send_email": {
      "email": {
        "to": "security-team@company.com",
        "subject": "SSH Bruteforce Detected",
        "body": "Multiple SSH failed login attempts detected from IP: {{ctx.payload.aggregations.source_ips.buckets.0.key}}"
      }
    }
  }
}

5. ELK Stack SIEM Implementation
5.1 Advanced ELK Configuration
Custom Logstash Pipelines
ruby
# logstash/pipeline/01-syslog.conf
input {
  tcp {
    port => 5000
    type => "syslog"
  }
  udp {
    port => 5000
    type => "syslog"
  }
}

filter {
  if [type] == "syslog" {
    grok {
      match => { "message" => "%{SYSLOGTIMESTAMP:syslog_timestamp} %{SYSLOGHOST:syslog_hostname} %{DATA:syslog_program}(?:\[%{POSINT:syslog_pid}\])?: %{GREEDYDATA:syslog_message}" }
    }
    date {
      match => [ "syslog_timestamp", "MMM  d HH:mm:ss", "MMM dd HH:mm:ss" ]
    }
  }
}

output {
  elasticsearch {
    hosts => ["elasticsearch:9200"]
    index => "syslog-%{+YYYY.MM.dd}"
  }
}
